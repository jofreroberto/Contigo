        function leerTexto(texto) {
            if (!texto || !texto.trim()) return;
            window.speechSynthesis.cancel();
            const mensaje = new SpeechSynthesisUtterance(texto);
            mensaje.lang = 'es-ES';
            mensaje.rate = parseFloat(document.getElementById('velocidadVoz')?.value || 1);

        function iniciarReconocimientoVoz() {
            if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
                alert('⚠️ Tu navegador no soporta reconocimiento de voz. Puedes seguir usando los botones con normalidad. 💛');
                return;
            }
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            reconocimientoVoz = new SpeechRecognition();
            reconocimientoVoz.lang = 'es-ES';
            reconocimientoVoz.continuous = false;
            reconocimientoVoz.interimResults = false;
            reconocimientoVoz.onstart = function() { alert('🎤 Te escucho... Dime qué necesitas 💛'); };
            reconocimientoVoz.onresult = function(event) {
                const transcripcion = event.results[0][0].transcript.toLowerCase();
                procesarComandoVoz(transcripcion);
            };
            reconocimientoVoz.onerror = function(event) {
                if (event.error === 'not-allowed') alert('🔇 Necesito permiso para escucharte. Activa el micrófono en tu navegador. 💛');
                else alert('🤔 No pude entenderte. ¿Puedes repetírmelo? Estoy aquí contigo. 💛');
            };
            reconocimientoVoz.start();
        }
